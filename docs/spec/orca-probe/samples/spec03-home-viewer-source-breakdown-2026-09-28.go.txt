package console

import (
	"context"
	"os"
	"testing"
	"time"

	"github.com/erranliu/SteerLine/internal/steerline"
	"github.com/erranliu/SteerLine/internal/steerline/viewer"
)

func TestRealIsolatedConsoleReadBreakdown(t *testing.T) {
	path := os.Getenv("STEERLINE_REAL_LATENCY_CONFIG")
	if path == "" {
		t.Skip("set STEERLINE_REAL_LATENCY_CONFIG for an isolated read-only source timing probe")
	}
	refs, err := LoadProjects(path)
	if err != nil || len(refs) != 1 {
		t.Fatalf("load exact viewer project: %+v %v", refs, err)
	}
	ref := refs[0]
	ctx := context.Background()
	for i := 0; i < 3; i++ {
		start := time.Now()
		connection := steerline.ConnectService(ctx, ref.Config)
		connectTime := time.Since(start)
		_ = connection
		start = time.Now()
		observation := viewer.NewFileSource(ref.Config).Snapshot()
		snapshotTime := time.Since(start)
		_ = observation
		start = time.Now()
		store, err := steerline.OpenNativeTaskReader(ctx, &ref.Config)
		if err != nil {
			t.Fatal(err)
		}
		openTime := time.Since(start)
		query := steerline.QueryService{Store: store, Project: ref.Identity}
		start = time.Now()
		if _, err := query.ListNativeTasks(ctx, "", 500); err != nil {
			t.Fatal(err)
		}
		listTime := time.Since(start)
		start = time.Now()
		if _, err := query.ReadyNativeTaskSnapshot(ctx); err != nil {
			t.Fatal(err)
		}
		readyTime := time.Since(start)
		start = time.Now()
		if _, err := store.Query(ctx, ref.Identity); err != nil {
			t.Fatal(err)
		}
		domainTime := time.Since(start)
		if err := store.Close(); err != nil {
			t.Fatal(err)
		}
		start = time.Now()
		actual := (FileQuerySource{}).ReadProject(ctx, ref, "91024")
		readTime := time.Since(start)
		if actual.Observation.Runtime == nil || actual.Domain == nil {
			t.Fatalf("read project lost current center projection: %+v", actual)
		}
		t.Logf("sample=%d connect=%s snapshot=%s open=%s list=%s ready=%s domain=%s full=%s revision=%d", i, connectTime, snapshotTime, openTime, listTime, readyTime, domainTime, readTime, actual.Observation.Runtime.Revision)
	}
}
