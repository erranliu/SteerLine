package integration
import("context";"encoding/json";"os";"testing"; app "github.com/erranliu/SteerLine/internal/app"; execution "github.com/erranliu/SteerLine/internal/execution"; backend "github.com/erranliu/SteerLine/internal/execution/backend")
func TestReview80ActualAppRequirementsClaim(t *testing.T){
 for _,explicit:=range []bool{true,false}{t.Run(map[bool]string{true:"explicit_selected_profile",false:"default_profile"}[explicit],func(t *testing.T){
 f:=newLifecycleProject(t,"local")
 raw,err:=os.ReadFile(f.configPath);require(t,err)
 var project map[string]any;require(t,json.Unmarshal(raw,&project))
 project["execution_requirements"]=backend.ExecutionRequirements{Tools:[]backend.ToolRequirement{{ID:"git"}}}
 raw,err=json.Marshal(project);require(t,err);require(t,os.WriteFile(f.configPath,raw,0600))
 f.config,err=app.LoadConfig(f.configPath);require(t,err);require(t,f.config.ResolveIdentity(context.Background()))
 evaluation,err:=f.config.EvaluateRequirements(context.Background());require(t,err)
 if len(evaluation.Tools)!=1||evaluation.Tools[0].Status!=backend.RequirementPresent {t.Fatalf("required git unavailable: %+v",evaluation)}
 createIntegrationTask(t,context.Background(),f,901)
 store,err:=app.OpenNativeTaskCommandStore(context.Background(),&f.config);require(t,err);defer store.Close()
 commands,err:=app.NewCommandService(store,f.config);require(t,err)
 graph,err:=store.TaskGraphSnapshot(context.Background(),f.config.Identity());require(t,err)
 options:=execution.TaskClaimOptions{TaskID:"901",RequestID:newID(),ExpectedTaskVersion:graph.ByID["901"].Version,ExpectedAuthorityVersion:graph.Authority.AuthorityVersion,ExpectedGraphRevision:graph.Authority.GraphRevision,RequestedBy:"operator",Explicit:true}
 if explicit{profiles:=f.config.AvailableExecutionProfiles();fixed:=profiles[0].Freeze(f.config.ExecutionDirectory,f.config.Identity());options.ExecutionProfile=&fixed}
 receipt,err:=commands.ClaimNativeTask(context.Background(),f.config.ExecutionConfig(),options)
 if err!=nil{t.Fatalf("actual loaded app with present git requirement cannot claim: %v",err)}
 if receipt.Status!="claim_reserved"{t.Fatalf("claim %+v",receipt)}
 })}
}
