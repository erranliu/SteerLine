package main
import("fmt";"os";"os/exec")
func main(){
 args:=os.Args[1:]; if len(args)<2||args[0]!="git-proxy" {os.Exit(2)}
 gitArgs:=append([]string(nil),args[2:]...)
 if len(gitArgs)>0&&gitArgs[0]=="fetch" {if remote:=os.Getenv("STEERLINE_FAKE_GIT_FETCH_REMOTE");remote!="" {for i:=1;i<len(gitArgs);i++ {if gitArgs[i]=="origin" {gitArgs[i]=remote;break}}}}
 cmd:=exec.Command(args[1],gitArgs...);cmd.Stdout=os.Stdout;cmd.Stderr=os.Stderr
 if err:=cmd.Run();err!=nil{if exit,ok:=err.(*exec.ExitError);ok{os.Exit(exit.ExitCode())};fmt.Fprintln(os.Stderr,err);os.Exit(1)}
}
